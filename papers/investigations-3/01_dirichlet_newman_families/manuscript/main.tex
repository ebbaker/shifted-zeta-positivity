\begin{abstract}
We give an explicit lower bound for the off-zero logarithmic derivative of
a primitive Dirichlet heat deformation. A scalar comparison then controls
the time at which all its zeros become real. A real sine tail added to
canonical-product truncations preserves the required initial lower bound
on the whole real line and allows the comparison to pass to the original
function without a compact-confinement hypothesis. In the normalization
$s=(1+iz)/2$, the argument gives $\Lambda_\chi\le27/100$ for every primitive
character of conductor $q\ge10^{12}$, and
$\Lambda_\chi=O((\log q)^{-4/7})$ uniformly as $q\to\infty$.
The finite scalar inequalities use exact rational enclosures. The analytic
proof details are included in the appendices; the exact scalar programme
and its output form the computational supplement. A sibling investigation
obtains the stronger finite bound $43/200$ using a uniform positive-time
floor and a restarted comparison; the present draft retains the simpler
initial-floor argument and its conductor-aspect rate.
\end{abstract}

\section{Normalization and statements}
Let $\chi$ be primitive of conductor $q>1$, with
$\chi(-1)=(-1)^{\mathfrak a}$, $\mathfrak a\in\{0,1\}$.
Choose a unit phase $\alpha_\chi$ with
$\alpha_\chi^2\varepsilon(\chi)=1$, where
$\varepsilon(\chi)=\tau(\chi)/(i^{\mathfrak a}\sqrt q)$, and set
\begin{align}
 \xi_\chi(s)&=\alpha_\chi(q/\pi)^{s/2}
 \Gamma\bigl((s+\mathfrak a)/2\bigr)L(s,\chi),\label{eq:xi}\\
 \Phi_\chi(u)&=4\alpha_\chi(\pi/q)^{\mathfrak a/2}
 e^{(1+2\mathfrak a)u}\sum_{n\ge1}\chi(n)n^{\mathfrak a}
 e^{-\pi n^2e^{4u}/q},\label{eq:phi}\\
 H_{t,\chi}(z)&=\int_{\mathbb R}e^{tu^2}\Phi_\chi(u)e^{izu}\,\dd u.
 \label{eq:heat}
\end{align}
Then $H_{0,\chi}(z)=\xi_\chi((1+iz)/2)$ and
$\partial_tH_{t,\chi}=-\partial_z^2H_{t,\chi}$.
The Hermitian functional equation makes $H_{t,\chi}$ real entire, although
it need not be even. With compatible phases,
$H_{t,\bar\chi}(z)=\overline{H_{t,\chi}(-\bar z)}$.
The normalization is detailed in Note~1 \cite{note1} and agrees with
Dobner's framework \cite{dobner}. Its associated constant $\Lambda_\chi$
is characterized by reality of all zeros exactly for $t\ge\Lambda_\chi$.

\begin{theorem}[Dirichlet consequence]\label{thm:dirichlet}
Every primitive Dirichlet character of conductor $q\ge10^{12}$ satisfies
\[\Lambda_\chi\le\frac{27}{100}.\]
Moreover, uniformly over primitive characters as $q\to\infty$,
\[0\le\Lambda_\chi=O\bigl((\log q)^{-4/7}\bigr),\qquad
\Lambda_\chi\longrightarrow0.\]
These statements use neither GRH, a quasi-RH half-plane, nor an assumed
confinement theorem for the character flow.
\end{theorem}

The lower bound is Dobner's published theorem \cite{dobner}. The upper
bound will follow from a general comparison. For $C>0$ define
\begin{align}
 D_C(t)&=1+2C(1-e^{-t/2}),&
 m_C(t)&=\frac{Ce^{-t/2}}{D_C(t)},\label{eq:dm}\\
 I_C(t)&=e^{-t}D_C(t)^{4/3},&
 J_C(t)&=\int_0^t I_C(s)\,\dd s.\label{eq:ij}
\end{align}

\begin{theorem}[Comparison without confinement]\label{thm:general}
Let $F$ be a nonzero real entire function of order at most one, with all
zeros in $|\Ima z|\le1$. Suppose
\[ -\Ima\frac{F'}F(x+3i)\ge C>0\qquad(x\in\mathbb R). \]
If $0<T\le1/2$ and $J_C(T)\ge1/2$, then
$e^{-T\partial_z^2}F$ has only real zeros.
\end{theorem}

The proof first treats flows with confined nonreal zeros, then removes
that restriction by a sine-tail approximation. No evenness assumption is
present. Appendices~\ref{app:characteristics}--\ref{app:heat} supply the analytic
details; Notes~2--3 \cite{note2,note3} retain their research history.

\section{Canonical sums and a positive-time lower bound}
For a real entire function with its zeros $\rho=a+iv$ in a bounded strip,
the genus-one product gives, at $p$ above that strip,
\begin{equation}
 S(x,p,t):=-\Ima\frac{F_t'}{F_t}(x+ip)
 =\sum_\rho\frac{p-v}{(x-a)^2+(p-v)^2}.\label{eq:poisson}
\end{equation}
Zeros are counted with multiplicity, including a possible zero at the
origin. The imaginary compensators cancel under conjugation. Their
absolute summability follows from bounded imaginary parts and
$\sum_{\rho\ne0}|\rho|^{-2}<\infty$. A separate, possibly divergent,
real sum of reciprocal zeros is never used.

For the Dirichlet flow, the double-exponential kernel decay gives
\begin{equation}
 \log\max_{|z|\le R}|H_{t,\chi}(z)|
 =O_{\chi,T}((1+R)\log(2+R))\quad(0\le t\le T).
 \label{eq:growth}
\end{equation}
The approximants in Section~\ref{sec:approx} have exponential-type growth.
These bounds justify the canonical differentiations and completeness
of the real characteristics below, as proved in
Appendix~\ref{app:characteristics}. No finite-type assumption is imposed
on the original $F$ of Theorem~\ref{thm:general}: this characteristic
argument is applied to its fixed approximants.

\begin{proposition}[Burgers lower bound]\label{prop:floor}
For the Dirichlet flow or the exponential-type approximants below,
suppose the zeros remain in $|\Ima z|\le1$ and $S(x,p,0)\ge C>0$.
Put $d=p-1>0$. Then
\begin{equation}
 S(x,p,t)\ge
 \frac{Ce^{-2t/d^2}}{1+dC(1-e^{-2t/d^2})}
 \qquad(x\in\mathbb R,\ 0\le t\le1/2).\label{eq:riccati}
\end{equation}
\end{proposition}
\begin{proof}
Write $U=F_t'/F_t=u-iS$ on the line of height $p$.
Equation~\eqref{eq:poisson} and its differentiated sums give
\[ |u_x|\le S/d,\qquad |S_{xx}|\le2S/d^2. \]
The heat equation implies $U_t=-U_{xx}-2UU_x$, hence
\[ S_t+2uS_x=-S_{xx}-2Su_x. \]
Along $x'=2u$, this yields $S'\ge-2S/d^2-2S^2/d$.
The Riccati solution with initial value $C$ is the right side of
\eqref{eq:riccati}. The characteristics are complete in both time
directions: growth and a uniformly zero-free anchor $ip$ imply
$|u(x,p,t)|=O((1+|x|)\log(2+|x|))$ on bounded time intervals, which
prevents finite-time escape. Their flow is onto $\mathbb R$, proving
uniformity in $x$.
\end{proof}

At $p=3$, the right side is $m_C(t)$ in \eqref{eq:dm}. This is a
quantified loss, rather than an assertion that the time-zero lower bound
persists unchanged.

\section{The comparison when contacts are compact}
We briefly retain the confinement hypothesis used in Note~2: on each
$[\delta,T]$, $\delta>0$, the nonreal zeros lie in a bounded horizontal
region. This hypothesis will be established for approximants only.

At a highest simple zero $x+ih$, put $Q=h^2$. The conjugate-pair kernel
\[ K_h(d,v)=\frac{h-v}{d^2+(h-v)^2}
                 +\frac{h+v}{d^2+(h+v)^2} \]
obeys $K_h/h\ge K_p/p$ whenever $|v|\le h$, $p^2\ge5h^2$,
and $(d,v)\ne(0,\pm h)$.
Clearing denominators reduces this to nonnegativity of
\[ D^2+(A+B-6r)D+(A-r)(B-r),\quad
D=d^2,\ A=h^2,\ B=p^2,\ 0\le r=v^2\le A. \]
The zero-motion identity, with the zero's own conjugate pair removed,
therefore gives
\begin{equation}
 Q'\le-2-\frac{4Q}{p}S(x,p,t)+\frac{8Q}{p^2-Q}.
 \label{eq:highest}
\end{equation}
This form of the comparison appears in Hakimoglu-Brown \cite{hb}; the
algebra and its non-even extension are recorded in Notes~1--2.

Classical strip contraction gives $Q\le1$. At $p=3$,
$8/(9-Q)\le1$, so Proposition~\ref{prop:floor} gives
\[ Q'\le-2-\left(\frac43m_C(t)-1\right)Q. \]
The corresponding solution from one is
\begin{equation}
 \overline Q(t)=\frac{1-2J_C(t)}{I_C(t)}.\label{eq:qbar}
\end{equation}
Until it reaches zero it decreases, with descent rate at least
$2-\overline Q\ge1$.

For completeness, use a strictly higher comparison curve, with initial
value $1+\varepsilon$ and a positive $\varepsilon$ added to its
differential equation. This supplies an initial time gap uniformly in
space. Confinement makes a later first contact compact. A simple contact
contradicts \eqref{eq:highest}; a multiple contact has a backward Hermite
split above the smooth curve. Letting $\varepsilon\downarrow0$ yields
$M(t)\le\overline Q(t)$ while the latter is positive. If it reaches zero
at $T_0$, strip contraction makes $F_{t+\overline Q(t)/2}$ real-rooted for
$t<T_0$. Heat continuity and Hurwitz then make $F_{T_0}$ real-rooted;
strip contraction preserves that property afterwards.
Lemma~\ref{lem:compact} below gives the compact-contact argument,
including the multiple-zero case and the endpoint.

\section{Removing confinement by a real sine tail}\label{sec:approx}
Fix $0<C_1<C$, where $C$ is the input in
Theorem~\ref{thm:general}, and write
\[F(z)=c e^{az}z^m\prod_{\rho\ne0}E_1(z/\rho),\qquad
 E_1(w)=(1-w)e^w,\quad c,a\in\mathbb R.\]
Let $P_R$ truncate this product at $|\rho|\le R$, retaining conjugate
pairs and multiplicities. Set
\begin{equation}
 \kappa=2(C_1+1),\quad h=\pi/\kappa,\quad M=\lfloor R/(4h)\rfloor,
 \qquad G_R(z)=P_R(z)\prod_{k>M}\left(1-\frac{z^2}{k^2h^2}\right).
 \label{eq:approx}
\end{equation}
The new zeros are real, and $G_R\to F$ locally uniformly.

\begin{lemma}\label{lem:approx}
For all sufficiently large $R$, $S_{G_R}(x,3,0)\ge C_1$ for every real
$x$. Each fixed approximant flow has uniformly confined nonreal zeros
on every bounded time interval. Moreover, $(G_R)_t\to F_t$ locally
uniformly, uniformly on bounded nonnegative time intervals.
\end{lemma}
\begin{proof}
We give the estimates here and prove the approximation and heat-limit
details in Appendix~\ref{app:heat}.
For $|x|\le R/2$, the omitted true-zero Poisson tail is at most
\[16(p+1)\sum_{|\rho|>R}|\rho|^{-2}=o(1).\]
For $|x|\ge R/2$, the full real lattice has lower bound
$\kappa\tanh(\kappa p)$, and removing its central zeros costs at most
$8p/(hR)+16p/R^2$. At $p=3$ this leaves a bound exceeding $C_1$;
the strict input gap $C-C_1$ handles the central region.

For fixed $R$, the sine product rewrites $G_R$ as
$e^{b_Rz}r_R(z)\sin(\kappa z)$ with $b_R$ real and $r_R$ real rational,
its poles canceled by sine zeros. If $r_R(z)\sim c_Rz^{d_R}$ at infinity,
Gaussian imaginary translation gives, uniformly for $|\Ima z|\le1$
and bounded $t$,
\begin{equation}
 \frac{(G_R)_t(z)}{c_Rz^{d_R}e^{b_Rz+(\kappa^2-b_R^2)t}}
 =\sin\bigl(\kappa(z-2b_Rt)\bigr)+O_{R,T}(|\Rea z|^{-1}).
 \label{eq:sine}
\end{equation}
Along the Gaussian line the real part remains $\Rea z$, so no rational
pole is crossed. Rouch\'e disks about the distant real sine zeros contain
exactly one zero each; reality forces it to be real. The complement in
the strip is zero-free. The strip itself follows from the genus-one
operator theorem of Cardon \cite[Theorem~1.5]{cardon}, with quadratic
parameter $2t$ and no quadratic exponential in $G_R$.

Finally, for any fixed $1<\sigma<2$, canonical-product bounds give
$|P_R(z)|\le e^{A(1+|z|)^\sigma}$ uniformly in $R$. The added tail is
bounded by $\sinh(\kappa|z|)/(\kappa|z|)\le e^{\kappa|z|}$.
These provide a common sub-Gaussian majorant in
\[ F_t(z)=\frac1{\sqrt\pi}\int_{\mathbb R}e^{-v^2}
                   F(z+2i\sqrt t\,v)\,\dd v. \]
Dominated convergence proves the heat-limit assertion. The limit is
nonzero: real Gaussian translation inverts imaginary Gaussian translation
under this same majorant, by Lemma~\ref{lem:heatlimit}.
\end{proof}

Apply the compact comparison to $G_R$. If $J_C(T)>1/2$, choose $C_1<C$
close enough that $J_{C_1}(T)>1/2$. The lemma makes $(G_R)_T$ real-rooted.
Hurwitz's theorem in each open half-plane passes this to the nonzero
limit $F_T$. For equality, first use $T+\eta<1/2$ and then let
$\eta\downarrow0$; the case $T=1/2$ is classical strip contraction.
This proves Theorem~\ref{thm:general}. No bound uniform in $R$ on the
approximants' confinement regions is required.

\section{Arithmetic input and exact scalar arithmetic}
The initial Dirichlet logarithmic derivative obeys
\begin{align}
 S_\chi(x,p,0)&=\frac12\Rea\frac{\xi_\chi'}{\xi_\chi}
                    \left(\frac{1+p+ix}{2}\right),\label{eq:initialidentity}\\
 S_\chi(x,p,0)&\ge C_{q,\mathfrak a}(p):=
 \frac14\left[\log\frac q\pi+
 \psi\left(\frac{1+p+2\mathfrak a}{4}\right)\right]
 -\frac12\left(-\frac{\zeta'}\zeta\right)\left(\frac{1+p}{2}\right).
 \label{eq:c}
\end{align}
Indeed, the Hermitian functional equation gives
\eqref{eq:initialidentity}; the gamma logarithmic derivative is bounded
below by its value on the real axis, and the Euler product gives
$|L'/L(s,\chi)|\le-\zeta'/\zeta(\Rea s)$ for $\Rea s>1$.
At $p=3$ the even case is the smaller of the two parities.

The finite certificate in Note~2, Section~5 \cite{note2}, proves
\begin{equation}
 C_{q,\mathfrak a}(3)>\frac{1221}{200}=6.105>6
 \qquad(q\ge10^{12},\ \mathfrak a=0,1).\label{eq:c6}
\end{equation}
It uses $\log10>23/10$, $\log4<7/5$, $\pi<4$,
$\gamma<H_{1000}-\log1000<29/50$, and a prime-power sum through 1000
with the positive tail $(\log1000+1)/1000$ to prove
$(-\zeta'/\zeta)(2)<3/5$. All decisions use exact rational enclosures.

The integral in \eqref{eq:ij} has the elementary form
\begin{equation}
 J_C(t)=\frac{3(1+2C)}{14C^2}(D_C(t)^{7/3}-1)
       -\frac3{20C^2}(D_C(t)^{10/3}-1).\label{eq:exactj}
\end{equation}
For $C=6$, $T=27/100$, the same certificate encloses
\[0.502464325582617834\le J_6(T)\le0.502464325582617835.\]
Alternating exponential series and rational cube-root bisection make
the lower endpoint rigorous. Theorem~\ref{thm:general} and
\eqref{eq:c6} prove the explicit part of Theorem~\ref{thm:dirichlet}.
The computational supplement consists of
\texttt{certify\_dirichlet\_scalar\_conditional\_20261017.py} and its JSON
output in the investigation's \texttt{numerics/} directory. It uses
Python's rational arithmetic; the deciding bounds and input parameters
are those just displayed. The build record pins their hashes. The
historical word ``conditional'' in their filenames records the original
analytic contract; Section~\ref{sec:approx} supplies it without changing
the arithmetic.

For $0\le t\le1/2$,
\[J_C(t)\ge e^{-1/2}
\frac{(1+Ce^{-1/4}t)^{7/3}-1}{(7/3)Ce^{-1/4}}.\]
Consequently
\begin{equation}
 B(C):=\frac{e^{1/4}}C
 \left[\left(1+\frac76e^{1/4}C\right)^{3/7}-1\right]
 \label{eq:bc}
\end{equation}
is a sufficient reality time whenever $B(C)\le1/2$ and
$C\le C_{q,\mathfrak a}(3)$. Since
$C_{q,\mathfrak a}(3)=\frac14\log q+O(1)$ uniformly in parity,
$B(C)=O((\log q)^{-4/7})$ for the natural input. This proves the
remaining assertions of Theorem~\ref{thm:dirichlet}.

\section{Scope, controls, and provenance}
The sine-tail limit bypasses confinement for the original Dirichlet
flow; it does not prove that separate assertion. A periodic real-entire
heat flow may have a positive initial logarithmic-derivative floor and
unboundedly many nonreal zeros at positive times. The example and its
scope are in Note~2, Section~7. This does not contradict the present
quantitative comparison.

The arithmetic gain enters through \eqref{eq:c}. Euler-factor collision
controls can violate its precise Euler-product allowance. Weighted-square,
reflected, and primitive-compatible controls need not have the character
completion or that allowance. The general comparison remains valid when
their actual input floor meets its assumptions; no character-specific
lower bound is assigned to them by analogy. This is smoothing at
positive time, not a proof of GRH at time zero.

A literature correction is necessary. Ki--Kim--Lee
\cite[Section~6, Example~1]{kkl} already extend their qualitative
strict bound below $1/2$ to real primitive Dirichlet characters. The
initial investigation's search missed that precedent. Their example
for a complex character uses a real self-dual combination, rather than
the individual non-even family in \eqref{eq:heat}. We make no priority
claim for the quantitative results of this draft.

\paragraph{Stronger sibling benchmark.}
Project~03's round-two notes \cite{p03floor,p03restart} prove in-note
$S_\chi(x,3,t)>6$ for all real $x$, $0\le t\le1/4$ and primitive
$q\ge10^{12}$. An intermediate-strip comparison, restarted on the
original flow, then gives $\Lambda_\chi\le43/200=0.215$.
The scalar checks use Arb enclosures and exact rational inequalities;

That result currently has internal same-model checks. The $0.27$ bound

\section{Growth and complete characteristics}\label{app:characteristics}

Here and below a bounded time interval means a compact interval of
nonnegative times. The Hermitian theta kernel in \eqref{eq:phi} satisfies
\[
 |\Phi_\chi(u)|\le K_\chi e^{3|u|}
                  \exp(-c_\chi e^{4|u|}),\qquad c_\chi>0.
\]
On the positive half-line this follows from its series, and on the
negative half-line from $\Phi_\chi(-u)=\overline{\Phi_\chi(u)}$, the
theta transformation in Dobner's normalization. Absorbing bounded
complex-time weights and $e^{3|u|}$ into half of the double exponential
bounds the integral on $|z|\le R$ by a constant times
\[
 1+\int_0^\infty e^{Ru-ce^{4u}}\,\dd u
 \le 1+\tfrac14c^{-R/4}\Gamma(R/4)\quad(R\ge1).
\]
Stirling's bound proves \eqref{eq:growth}. The same domination proves
joint analyticity and differentiation under the integral. Fourier
injectivity excludes the zero function. For each fixed approximant,
$|G_R(z)|\le K_R e^{A_R|z|}$ after enlarging $A_R$ to absorb its
polynomial factor. Gaussian translation gives this same form of bound
uniformly in time. Its heat flow is nonzero by Lemma~\ref{lem:heatlimit}.

\begin{lemma}[Complete characteristics]\label{lem:complete}
Suppose a nonzero real-entire heat flow $F_t$ has all zeros in
$|\Ima z|\le1$ and
$\log\max_{|z|\le R}|F_t(z)|\le K(1+R)\log(2+R)$ uniformly on a
bounded time interval. Then for $p>1$ its real characteristic equation
$x'=2\Rea(F_t'/F_t)(x+ip)$ has unique complete trajectories, forwards
and backwards on that interval. The map between any two times is onto
$\mathbb R$.
\end{lemma}
\begin{proof}
The zero-free point $ip$ is a uniform anchor:
$\inf_t|F_t(ip)|>0$ by continuity. Jensen's formula on disks centred
there, together with the growth bound, gives uniform zero counts
$N_t(R)=O(R\log(2+R))$. The logarithmic derivative at the anchor is
uniformly bounded. Retaining canonical summands until subtracting gives
the absolutely convergent identity
\[
 U_t(x+ip)-U_t(ip)=\sum_\rho
 \left(\frac1{x+ip-\rho}-\frac1{ip-\rho}\right),
 \qquad U_t=F_t'/F_t.
\]
The possible origin zero is included. Put $d=p-1$ and choose
$R\ge2(1+|x|+p)$. The terms $|\rho|\le2R$ contribute
$O(N_t(2R)/d)$. The others contribute
$O(R\sum_{|\rho|>2R}|\rho|^{-2})=O(\log(2+R))$, by the zero count
and partial summation. Consequently
\[
 |\Rea U_t(x+ip)|\le K_p(1+|x|)\log(2+|x|).
\]
The vector field is smooth, so local uniqueness holds. Since
$\int^\infty [(1+r)\log(2+r)]^{-1}\,\dd r=\infty$, this bound
precludes escape to infinity in finite time in either direction.
Solving backwards from every final point proves surjectivity.
\end{proof}

The canonical derivative sums converge locally uniformly off the strip.
In particular $U_x=-\sum_\rho(z-\rho)^{-2}$ and
$U_{xx}=2\sum_\rho(z-\rho)^{-3}$. At $z=x+ip$, each
$y=p-\Ima\rho\ge d$ and $|z-\rho|\ge y$, so
\[
 |U_x|\le\sum_\rho |z-\rho|^{-2}\le S/d,
 \qquad |\Ima U_{xx}|\le2\sum_\rho |z-\rho|^{-3}\le2S/d^2.
\]
These are the estimates used in Proposition~\ref{prop:floor}.
The preceding proof applies to the Dirichlet flow and to each fixed
approximant, with constants allowed to depend on that flow. It does not
assert the displayed growth bound for every entire function of order one.

\section{The compact-contact lemma}\label{app:compact}

\begin{lemma}\label{lem:compact}
Let $F_t$ be a jointly analytic, nonzero real-entire heat flow with
the uniform growth bound of Lemma~\ref{lem:complete}, initial strip
$|\Ima z|\le1$, classical strip contraction, and
$S(x,3,t)\ge m_C(t)$ for all real $x$. Suppose the nonreal zeros are
uniformly confined in the real direction on every $[\delta,T]$ with
$\delta>0$. Then, while \eqref{eq:qbar} is positive,
every zero satisfies $(\Ima z)^2\le\overline Q(t)$. If
$J_C(T)\ge1/2$, all zeros at time $T$ are real.
\end{lemma}
\begin{proof}
Set $a(t)=4m_C(t)/3-1$. For $\epsilon>0$, let
\[
 q_\epsilon'=-2-a(t)q_\epsilon+\epsilon,
 \qquad q_\epsilon(0)=1+\epsilon.
\]
On any compact interval before the first zero of $\overline Q$,
$q_\epsilon>\overline Q>0$ and $q_\epsilon\to\overline Q$
uniformly. Initially $q_\epsilon>1$ on a time interval independent
of position, whereas the actual squared heights never exceed one.
If a zero first reaches $q_\epsilon$, that contact therefore has
positive time. Confinement and the strip put the possible contacts in
a compact set. Continuity of zeros, or the argument principle on small
circles, then gives an attained first contact.

For a simple contact at $z=x+ih$, $h>0$, the implicit function theorem
and the heat equation give $z'=F_t''(z)/F_t'(z)$. In the canonical
product, the conjugate zero contributes $-1/h$ to $h'$. Summing over
the remaining zeros in the closed upper half-plane, with weight one
for a nonreal zero and half for a real zero, gives
\[
 h'=-\frac1h-2\sum{}'\varpi_k K_h(x-\Rea\rho_k,\Ima\rho_k).
\]
The real exponential drift contributes no imaginary part. The kernel
inequality of Section~3 gives \eqref{eq:highest}. At first contact the
zero is highest, $h^2=q_\epsilon\le1$, and hence
$(h^2)'\le-2-a(t)q_\epsilon<q_\epsilon'$, a contradiction.

For a contact of multiplicity $m\ge2$ at $(t_0,z_0)$, write
$F_{t_0}(z_0+w)=c w^m+O(w^{m+1})$, $c\ne0$. Joint analyticity and
the heat equation imply, locally uniformly in $\zeta$,
\[
 v^{-m/2}F_{t_0-v}(z_0+\sqrt v\,\zeta)
 \longrightarrow c\,e^{\partial_\zeta^2}\zeta^m
 =c\sum_{j=0}^{\lfloor m/2\rfloor}
       \frac{m!\,\zeta^{m-2j}}{j!(m-2j)!}.
\]
The last polynomial is a rescaled Hermite polynomial: its roots are
simple and purely imaginary, and for $m\ge2$ one has positive
imaginary part. Rouch\'e's theorem on a small circle about that root
therefore gives an earlier zero with height
$\Ima z_0+c_m\sqrt v+o(\sqrt v)$, $c_m>0$. The positive comparison
height $\sqrt{q_\epsilon(t)}$ changes by $O(v)$ near $t_0$.
This contradicts the absence of earlier contact. Thus no contact is
possible, and letting $\epsilon\downarrow0$ proves the strip bound.

Let $T_0$ be the first zero of $\overline Q$. For every $t<T_0$,
strip contraction makes $F_{t+\overline Q(t)/2}$ real-rooted.
Those times tend to $T_0$. Heat continuity and Hurwitz in each open
half-plane make the nonzero $F_{T_0}$ real-rooted, and strip contraction
preserves reality for all later times. Since $I_C>0$ and $J_C$ is
strictly increasing, $J_C(T)\ge1/2$ implies $T_0\le T$.
\end{proof}

\section{Sine tails and passage through the heat operator}\label{app:heat}

\subsection{The floor on the whole real line}
For $R$ large enough, $|x|\le R/2$ and $|\rho|>R$, the initial strip
gives $|x-\Rea\rho|\ge|\rho|/4$. Thus the omitted Poisson mass is
at most $16(p+1)\sum_{|\rho|>R}|\rho|^{-2}=o(1)$, uniformly on that
central region. All added zeros are real and contribute nonnegative
mass. The strict gap $C-C_1$ gives the required central floor.

The full lattice $h\mathbb Z$ has mass
\[
 \sum_{k\in\mathbb Z}\frac{p}{(x-kh)^2+p^2}
 =\frac{\kappa\sinh(2\kappa p)}
        {\cosh(2\kappa p)-\cos(2\kappa x)}
 \ge\kappa\tanh(\kappa p).
\]
For $|x|\ge R/2$ and $|k|\le M$, $|x-kh|\ge R/4$.
Removing these $2M+1\le R/(2h)+1$ terms costs at most
$8p/(hR)+16p/R^2$. At $p=3$,
$\kappa\tanh(3\kappa)>C_1$ because $\kappa=2(C_1+1)>2$ and
$\tanh6>1/2$. The remaining true zeros again contribute nonnegative
mass. This proves the global floor for large $R$.

\subsection{Confinement for a fixed approximant}
Euler's product for sine turns \eqref{eq:approx} into
$G_R(z)=e^{b_Rz}r_R(z)\sin(\kappa z)$, where $b_R$ is real and
$r_R$ is real rational with only real poles, canceled by sine zeros.
Write $r_R(z)\sim c_Rz^{d_R}$, where $d_R$ is an integer, possibly
negative. For $z=x+iy$, $|y|\le1$, $0\le t\le T$, and
$w=z+2i\sqrt t\,v$, rational asymptotics give, for sufficiently large
$|x|$,
\[
 \frac{r_R(w)}{c_Rz^{d_R}}
 =1+O_{R,T}\left(\frac{(1+|v|)^{K_R}}{|x|}\right).
\]
Indeed $\Rea w=x$ avoids every pole, $|w|\ge|x|$, and powers of
$w/z=1+2i\sqrt t\,v/z$ have this bound for positive or negative
integer exponents. The sine factor grows at most like
$e^{\kappa(1+2\sqrt T|v|)}$ along this line, while the extra
$e^{2ib_R\sqrt t\,v}$ has modulus one. Integrating against $e^{-v^2}$
therefore gives a uniform $O_{R,T}(|x|^{-1})$ error. Evaluating the
two elementary exponential Gaussian integrals in the leading term
proves \eqref{eq:sine}.

Cardon's theorem, applied to the genus-one entire $G_R$ with initial
strip one and operator $e^{-t\partial_z^2}$, confines all zeros to
$|\Ima z|\le\sqrt{\max(1-2t,0)}$. Choose a fixed disk radius
$0<r<\min(h/4,1/2)$ about each distant sine zero $kh+2b_Rt$.
On each boundary the leading sine has a positive lower modulus
independent of $k$ and $t$. Equation~\eqref{eq:sine} and Rouch\'e
give exactly one zero in every sufficiently distant disk. Conjugation
fixes each disk and the function is real entire, so that single zero
is real. On the remaining part of $|\Ima z|\le1$ the sine is bounded
away from zero, so the same estimate excludes other distant zeros.
The required cutoff may depend on $R$ and $T$. This proves uniform
confinement for each fixed approximant, even down to time zero.

\subsection{A common majorant, nonvanishing and the limit}
\begin{lemma}\label{lem:heatlimit}
The approximants converge under Gaussian heat translation locally
uniformly in $z$ and uniformly on bounded nonnegative time intervals.
Every heat translate of the original $F$ and of an approximant is
nonzero. The heat translations form a semigroup on these functions.
\end{lemma}
\begin{proof}
Fix $1<\sigma<2$. An order-at-most-one zero set has
$\sum_{\rho\ne0}|\rho|^{-\sigma}<\infty$. The elementary bound
$|E_1(w)|\le\exp(C_\sigma|w|^\sigma)$ follows from its quadratic
Taylor estimate for $|w|\le1$ and $|1-w|e^{|w|}$ for $|w|>1$.
Consequently, including $ce^{az}z^m$,
$|P_R(z)|\le\exp(A(1+|z|)^\sigma)$ with $A$ independent of $R$.
For the tail,
\[
 \left|\prod_{k>M}\left(1-\frac{z^2}{k^2h^2}\right)\right|
 \le\prod_{k\ge1}\left(1+\frac{|z|^2}{k^2h^2}\right)
 =\frac{\sinh(\kappa|z|)}{\kappa|z|}\le e^{\kappa|z|}.
\]
The omitted lattice tail tends to one locally uniformly, because
$\sum_{k>M}k^{-2}\to0$. Hence $G_R\to F$ locally uniformly and
all have a common sub-Gaussian majorant. In
\[
 (G_R)_t(z)=\pi^{-1/2}\int_{\mathbb R} e^{-v^2}
               G_R(z+2i\sqrt t\,v)\,\dd v
\]
this majorant is integrable uniformly for $z$ in a compact set and
$0\le t\le T$. Splitting into bounded $v$ and the uniform tail proves
the asserted convergence, including $t=0$.

For any such input $F$, Gaussian translation also preserves a bound
of order $\sigma<2$. Real translation inverts its imaginary heat
translation: absolute convergence permits Fubini in
\[
 \pi^{-1}\int_{\mathbb R^2}e^{-u^2-v^2}
       F(z+2\sqrt t\,u+2i\sqrt t\,v)\,\dd u\,\dd v=F(z).
\]
In polar coordinates the angular mean is $F(z)$ by the mean-value
identity for an entire function, and the radial integral has mass one.
Thus $F_t\equiv0$ would imply $F\equiv0$. Finally two imaginary
Gaussian translations combine by an orthogonal change of the two real
Gaussian variables, giving $(F_t)_s=F_{t+s}$. All exchanges are
justified by the same sub-Gaussian domination.
\end{proof}
