\section{Vandermonde and pair-matrix limits}\label{sec:geometry}
Choose any $m$ consecutive zeros from the $2N$ zeros in
Lemma~\ref{lem:carrier}. In source-mean coordinates write
\[
 y_i=\frac{\mu a_i}{\pi}=c+i+e_i,\qquad |e_i|\le e:=s/\pi<\tfrac12 .
\]
Let $\Delta_m(a)=\prod_{i<j}(a_j-a_i)^2$, the squared
Vandermonde, and define its explicit clock value by
\begin{equation}
 \Delta_m^{\rm clock}(L)=
 \left(\frac{\pi}{L}\right)^{m(m-1)}
       \prod_{k=1}^{m-1}(k!)^2 .
 \label{eq:Dclock}
\end{equation}
The distinction between the Vandermonde and its square fixes all
factors of two in the following estimate.

\begin{theorem}[Growing-block determinant comparison]\label{thm:V}
Let $e_s^*=\min\{e_\mu,e_s\}$ be the score allowance in
Lemma~\ref{lem:score}. If $L>e_s^*$, then
\begin{equation}
 \left|\log\frac{\Delta_m(a)}{\Delta_m^{\rm clock}(L)}\right|
 \le \mathcal V_m:=
 m(m-1)\frac{e_s^*}{L-e_s^*}
 +\frac{4e}{1-2e}\{mH_{m-1}-(m-1)\},
 \label{eq:V}
\end{equation}
where $H_{m-1}=\sum_{d=1}^{m-1}d^{-1}$ and $\mathcal V_1=0$.
For the choice \eqref{eq:growing}, uniformly for $m\le2N$,
\begin{equation}
 \mathcal V_m=O(L\kappa^{-1/3}
                    +L\log\kappa\,\kappa^{-1/6})=o(1).
 \label{eq:Vorder}
\end{equation}
Consequently $\Delta_m/\Delta_m^{\rm clock}(L)\to1$ uniformly
for these growing consecutive blocks.
\end{theorem}
\begin{proof}
For index distance $d$, the unfolded gap equals
$d(1+\theta_{ij})$ with $|\theta_{ij}|\le2e/d$.
Apply $|\log(1+\theta)|\le|\theta|/(1-|\theta|)$
and sum over the $m-d$ pairs of distance $d$, retaining the factor two
from the squared Vandermonde. This yields the second term of
\eqref{eq:V}. Since
$|\log(L/\mu)|\le e_s^*/(L-e_s^*)$, changing the reference spacing
from $\pi/\mu$ to $\pi/L$ costs the first term.
The identity $\prod_{d=1}^{m-1}d^{2(m-d)}
=\prod_{k=1}^{m-1}(k!)^2$ proves \eqref{eq:Dclock}.
Now $m=O(L\kappa^{1/3})$, $e=O(\kappa^{-1/2})$ and
$e_s^*=O(\kappa^{-1})$ imply \eqref{eq:Vorder}.
\end{proof}

Let $\mathsf L_{\rm pair}(a)$ be the graph Laplacian with edge weights
$(a_i-a_j)^{-2}$; equivalently
$\mathsf L_{\rm pair}=\tfrac12\operatorname{Hess}(-\log\Delta_m)$.
Let $\mathsf L_{\rm clock}$ have the weights $(i-j)^{-2}$.
Termwise comparison in the corresponding quadratic forms yields
\begin{equation}
 \left(\frac\mu L\right)^2(1+2e)^{-2}\mathsf L_{\rm clock}
 \preceq
 \left(\frac\pi L\right)^2\mathsf L_{\rm pair}(a)
 \preceq
 \left(\frac\mu L\right)^2(1-2e)^{-2}\mathsf L_{\rm clock}.
 \label{eq:Loewner}
\end{equation}
The comparison includes the constant zero mode and all nonzero
eigenvalues, uniformly in dimension.

The score estimate improves the first term of the earlier
$O(L\kappa^{-1/12}+L\log\kappa\,\kappa^{-1/6})$ bound
without changing its root-location term. These are slice estimates. For a selected block,
$H=U_0\prod_i(z-a_i)$ gives the complete zero motion
\begin{equation}
 a_i'=2\sum_{j\ne i}(a_i-a_j)^{-1}
                   +2(U_{0,z}/U_0)(a_i).
 \label{eq:exterior}
\end{equation}
Unfolding by $y_i=\gamma(t)a_i$ also adds
$(\gamma'/\gamma)y_i$.
The exterior force and this moving normalization cannot be discarded.
The determinant is translation invariant and does not determine a
density or all gaps by itself. We derive determinant closeness from
root-location estimates, not conversely.
The antecedent pair-matrix and transport notes
\cite{hdc06,hdc09,lhct07} explain why a selected theta block
does not inherit the unforced finite-polynomial equilibrium law.

\section{All-height scope and an analytic fallback}\label{sec:global}
\begin{proposition}[Fixed-regulator obstruction]\label{prop:obstruction}
For each fixed finite $t$, fixed $\epsilon>0$ and $Y<\infty$,
\begin{equation}
 \rh(t,x+iy)\longrightarrow0
 \quad\hbox{as }|x|\to\infty,\quad\hbox{uniformly for }|y|\le Y.
 \label{eq:vanishing}
\end{equation}
This remains true for positive normalized spatial averages whose
transverse support stays bounded and whose horizontal supports satisfy
$|x|-\ell(x,t)\to\infty$.
For that class, whenever $k_h(y)>0$,
\begin{equation}
 \sup_{x\in\R}\left|\frac{\pi}{L(t)}R(t,x,y)-k_h(y)\right|
 \ge k_h(y)
 \label{eq:obstruction}
\end{equation}
at each finite $t$ for which $L(t)$ is defined.
\end{proposition}
\begin{proof}
The source and its $u$ multiple are integrable with weight $e^{Yu}$.
Their Fourier transforms tend to zero, uniformly in bounded $y$.
For a direct uniform proof, first truncate in $u$ with a uniform
integrable majorant, then approximate the compact family of truncated
integrands in $L^1$ by finitely many step functions.
Their oscillatory integrals tend to zero.
Thus $H,H_z\to0$, and $\rh\le |H_z|^2/(\pi\epsilon^2)$ proves
\eqref{eq:vanishing}. The support condition transfers this statement
to the averages. Taking $|x|\to\infty$ in their discrepancy proves
\eqref{eq:obstruction}.
\end{proof}
This does not exclude a different outer profile, a changing regulator,
unbounded transverse scales, or nonlocal averages reaching the origin.


\subsection{Quantitative tails before the spatial regulator switch}
The fixed-regulator obstruction has the following quantitative companion
for the complete source \cite{n25}. The field estimate is valid on any
fixed compact time interval; transferring it to the regulated background
uses a slab ending at $64$.

\begin{lemma}[Complete-source contour jets]\label{lem:contour-jets}
Fix a compact $I=[t_a,t_b]$, $Y\ge0$, and $0<a\le\pi/16$. Put
\[
 d=\cos(4a),\quad C_a=(1-16e^{-3\pi d})^{-1},\quad
 s=\max(t_b,0),\quad D=s+Y+10,\quad V=1+\log(D/d).
\]
For each fixed integer $j\ge0$, define the finite constant
\begin{align}
 A_j={}&e^{-t_aa^2}(2\pi^2+3\pi)C_a\biggl[
 e^{sV^2+(9+Y)V}\frac{(V+a)^{j+1}-a^{j+1}}{j+1}\notag\\
 &\hspace{38mm}
 +\frac12\sum_{k=0}^j\binom jk a^{j-k}
                     \Gamma\left(\frac{k+1}{2}\right)\biggr].
 \label{eq:contour-constant}
\end{align}
Then on the full height axis
\begin{equation}
 |\partial_z^jH(t,x+iy)|\le A_je^{-a|x|}
 \qquad(t\in I,\ x\in\R,\ |y|\le Y).
 \label{eq:contour-jets}
\end{equation}
Constants depend on the fixed compact slab; no uniformity as $t_b$ tends
to infinity is asserted.
\end{lemma}
\begin{proof}
The theta series converges normally for $|\Im u|<\pi/8$.
Its even real reflection identity continues analytically throughout that
strip. Termwise absolute values on $u=v\pm ia$, $v\ge0$, give the
complete-mode estimate
\begin{align}
 |\Phi(v\pm ia)|
 &\le\sum_{n\ge1}(2\pi^2n^4e^{9v}+3\pi n^2e^{5v})
                              e^{-\pi d n^2e^{4v}}\notag\\
 &\le(2\pi^2+3\pi)C_a e^{9v-\pi d e^{4v}}.
 \label{eq:contour-source-envelope}
\end{align}
Here $d\ge1/\sqrt2$ and the successive positive-mode ratio is at most
$16e^{-3\pi d}<1$. Analytic even reflection supplies the corresponding
bound on the negative half-line. All theta modes remain in the bound.
Write \eqref{eq:H} as one half of the full-line Fourier integral, and
move the entire contour to $\R+ia$ for $x\ge0$, or $\R-ia$ for $x<0$.
On both vertical ends at $\Re u=\pm M$, the integrand, also after
multiplication by $(iu)^j$, is bounded by a fixed constant times a
polynomial times
\[
 \exp\{sM^2+YM+|x|a-\pi d e^{4M}\},
\]
which tends to zero for each $x$. Thus neither a half-line endpoint nor
an artificial cutoff boundary has been dropped.

The shift contributes $e^{-a|x|}$, and the other differentiated factors
on either half-line are at most
$e^{-t_aa^2}(v+a)^je^{sv^2+Yv}$. The factor one half cancels the two
half-line upper bounds. Split the remaining positive integral at $V$.
On $[0,V]$ discard the negative exponential and integrate $(v+a)^j$.
For $v\ge V$,
\[
 (s+1)v^2+(9+Y)v\le Dv^2\le\tfrac{\pi d}{2}e^{4v},
\]
since $e^{4v}/v^2$ increases there, $V\le D/d$, and the inequality
holds at $V=1+\log(D/d)$. The residual exponential is at most $e^{-v^2}$.
Expanding $(v+a)^j$ and extending this tail integral to zero gives
\eqref{eq:contour-constant}. The same majorants justify every fixed jet
and the directly used identity $H_t=-H_{zz}$; no asymptotic remainder is
differentiated.
\end{proof}

\begin{theorem}[Pre-switch observable tails]\label{thm:preswitch-tail}
Fix $I=[t_a,t_b]$ with finite $t_a\le t_b\le64$, fixed $\ell,h>0$
and an observation strip $|y|\le Y_0$. Apply
Lemma~\ref{lem:contour-jets} on the full transverse buffer
$Y=Y_0+h$, all real $x$, and use its constants $A_j$.
The exact global regulator is time-only here. Put
\begin{align*}
 e_*&=\frac{\eta\Phi(0)\sqrt\pi}
 {2(1+\max(t_a^2,t_b^2))^{1/4}},&
 b&=\sup_{t\in I}\frac{|t|}{1+t^2},\\
 E(x)&=e^{-a\max(|x|-\ell,0)},&
 G&=\frac{2A_0A_1}{e_*^2},\\
 L_x&=\|K_x\|_1,\quad L_y=\|K_y\|_1,&
 L_2&=\|\Delta K\|_1.
\end{align*}
The regulated measurement background is
\begin{equation}
 (R_0,J_0,U_0,W_0,Q_0)=(0,0,r(t),1,0).
 \label{eq:preswitch-background}
\end{equation}
Uniformly for $t\in I$, all real $x$ and $|y|\le Y_0$,
\begin{align}
 |R|&\le\frac{A_1^2}{\pi e_*^2}E^2,&
 |J|&\le\frac{A_1A_2}{\pi e_*^2}E^2,\notag\\
 |U-r|&\le\frac{A_0^2}{e_*^2}E^2,&
 |W-1|&\le\frac{A_0^2}{e_*^2}E^2,\notag\\
 |U_i|&\le GE^2\quad(i=x,y),&
 |U_t-r_t|&\le\frac{2A_0A_2+bA_0^2}{e_*^2}E^2.
 \label{eq:preswitch-linear-budgets}\\
 |K*\mathcal B_\epsilon-r_t|
 &\le\frac{bA_0^2}{e_*^2}E^2,&
 |K*\mathcal S_\epsilon|&\le\frac{bA_1^2}{\pi e_*^2}E^2,
 \label{eq:preswitch-forcing}\\
 |R_t|&\le\frac{2A_1A_3+bA_1^2}{\pi e_*^2}E^2
          +\frac{4A_1^2A_0A_2}{\pi e_*^4}E^4,
 \label{eq:preswitch-density-time}\\
 |Q|&\le4G^2E^4,\qquad
 |\Delta Q|\le[6L_2+4(L_x^2+L_y^2)]G^2E^4.
 \label{eq:preswitch-covariance}
\end{align}
Spatial derivatives of the linear filtered quantities are bounded by
convolving their corresponding pointwise allowances with
$|\partial^\alpha K|$, through weak order four. All initial/final time
traces and spatial normal traces inherit the full-buffer bounds.
Spatial regulator curvature is identically zero; its time forcing and
current source are retained.
\end{theorem}
\begin{proof}
Every convolution point has
$e^{-a|\Re z|}\le E(x)$ and lies in $|\Im z|\le Y$.
Use $|H|^2+q\ge q$, $\log(1+|H|^2/q)\le|H|^2/q$ and
$|S_i|\le2|H||H_z|/q$ to obtain the static bounds.
The directly computed physical identities
\[
 S_t=-2\Re\frac{H_{zz}\overline H}{|H|^2+q}+r_tw,
 \qquad
 \rho_t=-\frac{2q\Re(H_{zzz}\overline{H_z})}{\pi(|H|^2+q)^2}
  +r_t(1-2w)\rho+4\rho\Re\frac{H_{zz}\overline H}{|H|^2+q}
\]
prove \eqref{eq:preswitch-forcing}--\eqref{eq:preswitch-density-time};
these are estimates on actual time derivatives.
For covariance retain its actual definition: the two mean squares and
two squared means have $|S_i|\le GE^2$.
Move derivatives of the first squares to $K$. For $v_i=K*S_i$ use
$\Delta(v_i^2)=2v_i\Delta v_i+2|\nabla v_i|^2$.
The respective Laplacian coefficients are $2L_2$ and
$4(L_2+L_x^2+L_y^2)$, giving
\eqref{eq:preswitch-covariance}. The fixed kernel's integrable weak
derivatives pay the remaining linear derivatives and traces. The
background \eqref{eq:preswitch-background} itself satisfies the exact
adaptive equations on this slab.
\end{proof}

For example a prescribed density tolerance $\delta>0$ is guaranteed on
the whole remaining tail by the explicit threshold
\begin{equation}
 |x|\ge\ell+\frac1{2a}\max\left\{0,
              \log\frac{A_1^2}{\pi e_*^2\delta}\right\}.
 \label{eq:preswitch-tail-threshold}
\end{equation}
This is an absolute statement about regulated observables when the
regulator dominates the complete field. It assigns no zero-counting
measure to the identically zero reference field and excludes no high
zeros. The canonical even reflection and complete coefficient envelope
are the premises of this contour application; atoms and heat polynomials
lack them, and changed-source controls do not inherit its constants.
Smooth colliding sources with suitable strip bounds can have analogous
tails. No joint large-time conclusion follows from constants on a fixed
compact slab.

For $t>64$, Section~\ref{sec:protocol} gives
$\epsilon_g=E_0e^{-\beta x^2/2}$ with $\beta>0$.
The smallest regulator on the horizontal buffer is
$E_0e^{-\beta(|x|+\ell)^2/2}$. The same absolute-upper-bound/floor
argument then pays a factor
$\exp\{\beta(|x|+\ell)^2-2a(|x|-\ell)\}$, which diverges.
This identifies failure of that payment, rather than divergence of the
true density. Proposition~\ref{prop:cutoff-potential} gives a distinct
failure of a finite-source representation for the potential.

\subsection{A fixed-regulator analytic hierarchy}

An absolute all-height approximation is nonetheless available.
For $t\ge0$, define
\begin{equation}
 P_N(t,z)=\frac18\sum_{k=0}^N\frac{(t/4)^k}{k!}
                   \xi^{(2k)}(\tfrac12+\tfrac i2z).
 \label{eq:PN}
\end{equation}
This is a finite analytic expression and uses no zero locations.

\begin{theorem}[A global analytic hierarchy]\label{thm:global}
For $s,Y\ge0$ put $V=1+\log(s+Y+10)$ and, for $j=0,1$,
\begin{equation}
 \mathcal B_j(s,Y)=2\pi^2C_b
 \left\{\frac{V^{j+1}}{j+1}e^{sV^2+(9+Y)V}
                    +\frac12\Gamma((j+1)/2)\right\}.
 \label{eq:majorant}
\end{equation}
For every positive normalized spatial filter at time $t$, supported in $|\Im z|\le Y$,
\begin{equation}
 |F\rh[H]-F\rh[P_N]|
 \le 2^{-N-1}\mathcal A(t,Y,\epsilon), \label{eq:globalbudget}
\end{equation}
where
\begin{equation}
 \mathcal A=
 \frac{2\mathcal B_1(t,Y)\mathcal B_1(2t,Y)}{\pi\epsilon^2}
 +\frac{c_0\mathcal B_1(t,Y)^2\mathcal B_0(2t,Y)}{\pi\epsilon^3},
 \qquad c_0=\frac{125}{54\sqrt5}.
 \label{eq:Aglobal}
\end{equation}
The bound is uniform over all real heights.
For prescribed $p>0$, the degree
\[
 N=\max\{0,\lceil\log_2\mathcal A+p\log_2(1+t)\rceil-1\}
\]
gives error at most $(1+t)^{-p}$ and is
$O_{Y,\epsilon,p}(t\log^2(t+10))$.
\end{theorem}
\begin{proof}
The source estimate in \eqref{eq:q} implies
$\int_0^\infty u^je^{su^2+Yu}\Phi(u)\,du\le\mathcal B_j(s,Y)$.
On $[0,V]$ discard the negative exponential.
For $u\ge V$, with $D_0=s+Y+10$, one has
$(s+1)u^2+(9+Y)u\le D_0u^2\le(\pi/2)e^{4u}$.
The last inequality follows at $V$ from $V\le D_0$ and then from
monotonicity of $e^{4u}/u^2$.
The integrand exponent is therefore at most $-u^2$, proving
\eqref{eq:majorant}.

Expanding $e^{tu^2}$ in the source integral gives \eqref{eq:PN}.
For $a\ge0$ its tail satisfies
$\sum_{k>N}a^k/k!\le2^{-N-1}e^{2a}$.
Consequently, on the entire strip,
\[
 |\partial_z^j(H-P_N)|\le2^{-N-1}\mathcal B_j(2t,Y),\qquad
 |H_z|,|(P_N)_z|\le\mathcal B_1(t,Y).
\]
The coefficient $\epsilon^2/\{\pi(|h|^2+\epsilon^2)^2\}$
has global gradient maximum $c_0/(\pi\epsilon^3)$:
maximize $4r\epsilon^2/(r^2+\epsilon^2)^3$ at
$r=\epsilon/\sqrt5$. A mean-value bound for the density gives
\eqref{eq:globalbudget}, and positivity transfers it to the filter.
The stated choice of $N$ then follows.
\end{proof}
A time average uses the average of the right side of \eqref{eq:globalbudget}
at its constituent times. This hierarchy is conservative: evaluation errors in the xi derivatives
and the filter must be added separately. It is a constructive absolute
budget, not an efficient outer-height asymptotic model.
